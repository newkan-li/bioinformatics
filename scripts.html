# Needleman-Wunsch 全局比对核心递归
# S(i,j)=max{S(i-1,j-1)+p(ai,bj), S(i-1,j)+p(ai,-), S(i,j-1)+p(-,bj)}
# 时间复杂度 O(mn)，空间复杂度 O(mn)

def global_align(a, b, match=1, mismatch=-1, gap=-2):
    m, n = len(a), len(b)
    S = [[0]*(n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        S[i][0] = S[i-1][0] + gap
    for j in range(1, n+1):
        S[0][j] = S[0][j-1] + gap
    for i in range(1, m+1):
        for j in range(1, n+1):
            diag = S[i-1][j-1] + (match if a[i-1] == b[j-1] else mismatch)
            up = S[i-1][j] + gap
            left = S[i][j-1] + gap
            S[i][j] = max(diag, up, left)
    return S[m][n]

print(global_align("ACGGTT", "ACGTT"))

# Needleman-Wunsch 全局比对（伪代码）
def needleman_wunsch(s, t, match=2, mismatch=-1, gap=-1):
    m, n = len(s), len(t)
    M = [[0]*(n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        M[i][0] = M[i-1][0] + gap
    for j in range(1, n+1):
        M[0][j] = M[0][j-1] + gap
    for i in range(1, m+1):
        for j in range(1, n+1):
            score = match if s[i-1] == t[j-1] else mismatch
            M[i][j] = max(
                M[i-1][j-1] + score,
                M[i-1][j] + gap,
                M[i][j-1] + gap
            )
    return M[m][n]

# 最大似然估计（正态分布均值和方差）
def neg_log_likelihood(params, y):
    mu, sigma = params
    if sigma <= 0:
        return np.inf
    n = len(y)
    return 0.5 * n * np.log(2 * np.pi * sigma ** 2) + np.sum((y - mu) ** 2) / (2 * sigma ** 2)

res = minimize(neg_log_likelihood, x0=[0, 1], args=(y,), method='Nelder-Mead')
print('MLE mu, sigma:', res.x)

# 系统聚类：最短距离法核心步骤
import numpy as np

def single_linkage(D, labels):
    D = D.astype(float).copy()
    np.fill_diagonal(D, np.inf)
    clusters = {i: [i] for i in range(len(labels))}
    active = set(clusters)
    while len(active) > 1:
        # 找当前最小类间距离
        best = None
        for i in active:
            for j in active:
                if i < j and (best is None or D[i, j] < best[0]):
                    best = (D[i, j], i, j)
        _, i, j = best
        # 合并 i, j 为新类 i
        clusters[i] += clusters[j]
        active.remove(j)
        # 最短距离更新：D[k, i] = min(D[k, i], D[k, j])
        for k in active:
            if k != i:
                D[i, k] = D[k, i] = min(D[i, k], D[j, k])
    return clusters

import numpy as np
from scipy.stats import f

def discriminant_f_test(nA, nB, p, between_var, within_var, alpha=0.05):
    """
    nA, nB: 两类样本数
    p: 变量数
    between_var: 组间差异
    within_var: 组内差异

    F = between_var / within_var
    df1 = p

# Viterbi algorithm for HMM decoding
# Input: HMM lambda=(A,B,pi), observation sequence O[1..T]
# Output: best state sequence Q[1..T]

def viterbi(A, B, pi, O):
    N = len(A)          # number of states
    T = len(O)          # length of observation sequence
    delta = [[0.0]*N for _ in range(T)]
    psi   = [[0]*N for _ in range(T)]

    # Initialization
    for i in range(N):
        delta[0][i] = pi[i] * B[i][O[0]]
        psi[0][i] = 0

    # Recursion
    for t in range(1, T):
        for j in range(N):
            best, arg = -1.0, 0
            for i in range(N):
                p = delta[t-1][i] * A[i][j]
                if p > best:
                    best, arg = p, i
            delta[t][j] = best * B[j][O[t]]
            psi[t][j]   = arg

    # Termination
    best, qT = -1.0, 0
    for i in range(N):
        if delta[T-1][i] > best:
            best, qT = delta[T-1][i], i

    # Backtracking
    Q = [0]*T
    Q[T-1] = qT
    for t in range(T-2, -1, -1):
        Q[t] = psi[t+1][Q[t+1]]
    return Q, best

# Time complexity: O(T * N^2)

# Elman (simple recurrent) network - conceptual PyTorch sketch
import torch
import torch.nn as nn

class ElmanNet(nn.Module):
    def __init__(self, in_dim, hidden_dim, out_dim):
        super().__init__()
        self.hidden_dim = hidden_dim
        self.fc_in  = nn.Linear(in_dim, hidden_dim)
        self.fc_ctx = nn.Linear(hidden_dim, hidden_dim, bias=False)
        self.fc_out = nn.Linear(hidden_dim, out_dim)
        self.act    = nn.Tanh()

    def forward(self, x, h=None):
        # x: (batch, in_dim); h: (batch, hidden_dim)
        if h is None:
            h = torch.zeros(x.size(0), self.hidden_dim)
        h = self.act(self.fc_in(x) + self.fc_ctx(h))
        y = self.fc_out(h)
        return y, h

```python
# SVM 分类决策函数核心形式
# 线性可分：M(x)=sgn(w*·x+b*)=sgn(Σ α_i* y_i (x_i·x)+b*)
# 线性不可分：M(x)=sgn(Σ α_i* y_i K(x_i,x)+b*)

import numpy as np

def svm_decision(X_sv, y_sv, alpha, b, x, kernel):
    """
    X_sv: 支持向量, shape=(n_sv, d)
    y_sv: 支持向量标签, shape=(n_sv,)
    alpha: 拉格朗日乘子, shape=(n_sv,)
    b: 偏置
    x: 待预测样本, shape=(d,)
    kernel: 核函数 K(a,b)
    """
    s = 0.0
    for xi, yi, ai in zip(X_sv, y_sv, alpha):
        s += ai * yi * kernel(xi, x)
    return np.sign(s + b)
```

```text
块算法（Chunking Algorithm）核心步骤：

输入：训练集 {(x_i, y_i)}，核函数 K，精度 ε
输出：支持向量及对应 α, b

1. 初始化工作集 W 为任意小规模子集
2. repeat
3.     在 W 上求解对偶二次规划子问题，得到 α_W
4.     保留 α_i > 0 的样本作为候选支持向量
5.     将违反 KKT 条件最严重的样本加入 W
6.     移除 α_i = 0 且非支持向量的样本
7. until 所有样本满足 KKT 条件或达到精度 ε
8. 输出最终支持向量、α、b

存储核矩阵在 n>4000 时约需 128 MB。
```

def decode_solid(colors, start_base):
    # colors: 颜色序列，如 ['R','B','G',...]
    # start_base: 已知起始碱基，如 'A'
    base_map = {
        ('A','A'):'R', ('A','C'):'B', ('A','G'):'G', ('A','T'):'Y',
        ('C','A'):'B', ('C','C'):'R', ('C','G'):'Y', ('C','T'):'G',
        ('G','A'):'G', ('G','C'):'Y', ('G','G'):'R', ('G','T'):'B',
        ('T','A'):'Y', ('T','C'):'G', ('T','G'):'B', ('T','T'):'R'
    }
    bases = [start_base]
    for color in colors:
        prev = bases[-1]
        for (b1,b2), c in base_map.items():
            if b1 == prev and c == color:
                bases.append(b2)
                break
    return ''.join(bases)

    # 谱聚类
    sc = SpectralClustering(n_clusters=n_clusters, affinity='precomputed')
    labels = sc.fit_predict(norm)
    return labels

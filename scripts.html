def edit_ops(a, b):
    if a == b:

全局比对与局部比对递归算法 p42
# Needleman-Wunsch 全局比对核心递归
# S(i,j)=max{S(i-1,j-1)+p(ai,bj), S(i-1,j)+p(ai,-), S(i,j-1)+p(-,bj)}
# 时间复杂度 O(mn)，空间复杂度 O(mn)

def global_align(a, b, match=1, mismatch=-1, gap=-2):
    m, n = len(a), len(b)
    S = [[0]*(n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        S[i][0] = S[i-1][0] + gap
    for j in range(1, n+1):
        S[0][j] = S[0][j-1] + gap
    for i in range(1, m+1):
        for j in range(1, n+1):
            diag = S[i-1][j-1] + (match if a[i-1] == b[j-1] else mismatch)
            up = S[i-1][j] + gap
            left = S[i][j-1] + gap
            S[i][j] = max(diag, up, left)
    return S[m][n]

print(global_align("ACGGTT", "ACGTT"))

def affine_gap_penalty(k, a=-4, b=-3):
    # k: 空位长度; a: 起始罚分; b: 延伸罚分
    return a + b * k

# 时间复杂度: O(1)

# dotplot 算法核心步骤（伪代码）
# 1. 构建 m x n 矩阵，m=|s|, n=|t|
# 2. 若 s[i] == t[j]，则 M[i][j] = 1，否则为 0

def dotplot(s, t):
    m, n = len(s), len(t)
    M = [[1 if s[i] == t[j] else 0 for j in range(n)] for i in range(m)]
    return M

# Needleman-Wunsch 全局比对（伪代码）
def needleman_wunsch(s, t, match=2, mismatch=-1, gap=-1):
    m, n = len(s), len(t)
    M = [[0]*(n+1) for _ in range(m+1)]
    for i in range(1, m+1):
        M[i][0] = M[i-1][0] + gap
    for j in range(1, n+1):
        M[0][j] = M[0][j-1] + gap
    for i in range(1, m+1):
        for j in range(1, n+1):
            score = match if s[i-1] == t[j-1] else mismatch
            M[i][j] = max(
                M[i-1][j-1] + score,
                M[i-1][j] + gap,
                M[i][j-1] + gap
            )
    return M[m][n]

# 时间复杂度: O(m*n)，空间复杂度: O(m*n)

    for j in range(1, n+1):
        M[0][j] = M[0][j-1] + gap
    for i in range(1, m+1):
        for j in range(1, n+1):
            d = match if s[i-1] == t[j-1] else mismatch
            M[i][j] = max(M[i-1][j-1]+d, M[i-1][j]+gap, M[i][j-1]+gap)
    return M

M = dp_global(s, t)
for row in M:
    print(row)
print("全局比对得分:", M[len(s)][len(t)])

# BLAST 核心步骤伪代码
# 输入: query, db, word_len=3, T=11, X=1
# 输出: 最佳局部比对 hits

def blast(query, db, word_len=3, T=11, X=1):
    words = make_words(query, word_len)      # 1) 编译查询序列生成词列表
    hits = []
    for rec in db:                            # 2) 扫描数据库
        for w in words:
            for pos in find_all(rec, w):
                hit = extend(rec, pos, query, T, X)  # 3) 向两端延伸
                if hit.score >= T:
                    hits.append(hit)
    return sorted(hits, key=lambda h: -h.score)

# 双序列 DP 为二维, 多序列 DP 为 N 维

# 伪代码 (N 条序列, 长度 L)
# score = zeros(L1+1, L2+1, ..., LN+1)
# for i1 in 0..L1:
#   for i2 in 0..L2:
#     ...

# 最小二乘估计与最大似然估计
import numpy as np
from scipy.optimize import minimize

# 最小二乘估计总体均值 mu
y = np.array([2.1, 1.9, 2.5, 2.0, 1.8])
mu_ls = np.mean(y)
print('LS mean:', mu_ls)

# 最大似然估计（正态分布均值和方差）
def neg_log_likelihood(params, y):
    mu, sigma = params
    if sigma <= 0:
        return np.inf
    n = len(y)
    return 0.5 * n * np.log(2 * np.pi * sigma ** 2) + np.sum((y - mu) ** 2) / (2 * sigma ** 2)

res = minimize(neg_log_likelihood, x0=[0, 1], args=(y,), method='Nelder-Mead')
print('MLE mu, sigma:', res.x)

# 系统聚类：最短距离法核心步骤
import numpy as np

def single_linkage(D, labels):
    D = D.astype(float).copy()
    np.fill_diagonal(D, np.inf)
    clusters = {i: [i] for i in range(len(labels))}
    active = set(clusters)
    while len(active) > 1:
        # 找当前最小类间距离
        best = None
        for i in active:
            for j in active:
                if i < j and (best is None or D[i, j] < best[0]):
                    best = (D[i, j], i, j)
        _, i, j = best
        # 合并 i, j 为新类 i
        clusters[i] += clusters[j]
        active.remove(j)
        # 最短距离更新：D[k, i] = min(D[k, i], D[k, j])
        for k in active:
            if k != i:
                D[i, k] = D[k, i] = min(D[i, k], D[j, k])
    return clusters

# 示例距离矩阵
D = np.array([[0, 1, 4],
              [1, 0, 2],
              [4, 2, 0]])

# 时间复杂度：朴素实现 O(n^3)，优化后可 O(n^2 log n)

# Fisher 两类线性判别：核心计算
import numpy as np

# A 组、B 组样本，每行一个样本，每列一个判别指标
A = np.array([[1.0, 2.0],
              [1.5, 1.8],
              [1.2, 2.2]])
B = np.array([[4.0, 5.0],
              [4.5, 4.8],
              [3.8, 5.2]])

muA = A.mean(axis=0)
muB = B.mean(axis=0)

# 组内离差阵
SA = ((A - muA).T @ (A - muA))
SB = ((B - muB).T @ (B - muB))
Sw = SA + SB

# Fisher 判别系数 c = Sw^{-1}(muA - muB)
c = np.linalg.solve(Sw, muA - muB)
print(c)

# 判别得分
yA = A @ c
yB = B @ c
print(yA.mean(), yB.mean())

# 判别函数显著性 F 检验（示意）
import numpy as np
from scipy.stats import f

def discriminant_f_test(nA, nB, p, between_var, within_var, alpha=0.05):
    """
    nA, nB: 两类样本数
    p: 变量数
    between_var: 组间差异
    within_var: 组内差异
    """
    F = between_var / within_var
    df1 = p
    df2 = nA + nB - p - 1
    F_crit = f.ppf(1 - alpha, df1, df2)
    return F, F_crit, F > F_crit

# 极简示例

    nA=20, nB=20, p=3, between_var=12.5, within_var=2.5

# Viterbi algorithm for HMM decoding
# Input: HMM lambda=(A,B,pi), observation sequence O[1..T]
# Output: best state sequence Q[1..T]

def viterbi(A, B, pi, O):
    N = len(A)          # number of states
    T = len(O)          # length of observation sequence
    delta = [[0.0]*N for _ in range(T)]
    psi   = [[0]*N for _ in range(T)]

    # Initialization
    for i in range(N):
        delta[0][i] = pi[i] * B[i][O[0]]
        psi[0][i] = 0

    # Recursion
    for t in range(1, T):
        for j in range(N):
            best, arg = -1.0, 0
            for i in range(N):
                p = delta[t-1][i] * A[i][j]
                if p > best:
                    best, arg = p, i
            delta[t][j] = best * B[j][O[t]]
            psi[t][j]   = arg

    # Termination
    best, qT = -1.0, 0
    for i in range(N):
        if delta[T-1][i] > best:
            best, qT = delta[T-1][i], i

    # Backtracking
    Q = [0]*T
    Q[T-1] = qT
    for t in range(T-2, -1, -1):
        Q[t] = psi[t+1][Q[t+1]]
    return Q, best

# Time complexity: O(T * N^2)
# Minimal example
A = [[0.7,0.3],[0.4,0.6]]
B = [[0.5,0.5],[0.1,0.9]]
pi = [0.6,0.4]
O = [0,1,0]

# Elman (simple recurrent) network - conceptual PyTorch sketch
import torch
import torch.nn as nn

class ElmanNet(nn.Module):
    def __init__(self, in_dim, hidden_dim, out_dim):
        super().__init__()
        self.hidden_dim = hidden_dim
        self.fc_in  = nn.Linear(in_dim, hidden_dim)
        self.fc_ctx = nn.Linear(hidden_dim, hidden_dim, bias=False)
        self.fc_out = nn.Linear(hidden_dim, out_dim)
        self.act    = nn.Tanh()

    def forward(self, x, h=None):
        # x: (batch, in_dim); h: (batch, hidden_dim)
        if h is None:
            h = torch.zeros(x.size(0), self.hidden_dim)
        h = self.act(self.fc_in(x) + self.fc_ctx(h))
        y = self.fc_out(h)
        return y, h

# Motivation: HMM assumes independent emissions;
# Elman nets capture temporal correlation (e.g. runs of
# hydrophobic residues buried in protein cores).
# Time complexity per step: O(in*h + h*h + h*out)

支持向量分类决策函数与支持向量回归 p283
```python
# SVM 分类决策函数核心形式
# 线性可分：M(x)=sgn(w*·x+b*)=sgn(Σ α_i* y_i (x_i·x)+b*)
# 线性不可分：M(x)=sgn(Σ α_i* y_i K(x_i,x)+b*)

import numpy as np

def svm_decision(X_sv, y_sv, alpha, b, x, kernel):
    """
    X_sv: 支持向量, shape=(n_sv, d)
    y_sv: 支持向量标签, shape=(n_sv,)
    alpha: 拉格朗日乘子, shape=(n_sv,)
    b: 偏置
    x: 待预测样本, shape=(d,)
    kernel: 核函数 K(a,b)
    """
    s = 0.0
    for xi, yi, ai in zip(X_sv, y_sv, alpha):
        s += ai * yi * kernel(xi, x)
    return np.sign(s + b)

# 时间复杂度：预测 O(n_sv * d)，n_sv为支持向量数，d为特征维数
```

SVM训练算法：块算法与固定工作样本集 p284
```text
块算法（Chunking Algorithm）核心步骤：

输入：训练集 {(x_i, y_i)}，核函数 K，精度 ε
输出：支持向量及对应 α, b

1. 初始化工作集 W 为任意小规模子集
2. repeat
3.     在 W 上求解对偶二次规划子问题，得到 α_W
4.     保留 α_i > 0 的样本作为候选支持向量
5.     将违反 KKT 条件最严重的样本加入 W
6.     移除 α_i = 0 且非支持向量的样本
7. until 所有样本满足 KKT 条件或达到精度 ε
8. 输出最终支持向量、α、b

时间复杂度：最坏约 O(n^2) ~ O(n^3)，n为训练样本数；
存储核矩阵在 n>4000 时约需 128 MB。

极简示例：
样本数 n=5，初始 W={x1,x2}；
求解后 x1 为支持向量，x2 的 α=0；
加入违反 KKT 的 x3，移除 x2；
重复至全部满足 KKT。
```

    signals = []

    for base in flow_order:

        while pos < len(template) and template[pos] == base:

# 双碱基编码矩阵：4种颜色对应16种双碱基组合
# 已知任一位置碱基，可逐步解码

def decode_solid(colors, start_base):
    # colors: 颜色序列，如 ['R','B','G',...]
    # start_base: 已知起始碱基，如 'A'
    base_map = {
        ('A','A'):'R', ('A','C'):'B', ('A','G'):'G', ('A','T'):'Y',
        ('C','A'):'B', ('C','C'):'R', ('C','G'):'Y', ('C','T'):'G',
        ('G','A'):'G', ('G','C'):'Y', ('G','G'):'R', ('G','T'):'B',
        ('T','A'):'Y', ('T','C'):'G', ('T','G'):'B', ('T','T'):'R'
    }
    bases = [start_base]
    for color in colors:
        prev = bases[-1]
        for (b1,b2), c in base_map.items():
            if b1 == prev and c == color:
                bases.append(b2)
                break
    return ''.join(bases)

# 重硫酸盐测序数据分析核心步骤（Python伪代码）
# 1. 重硫酸盐处理：未甲基化C->U，甲基化C不变
# 2. 测序后比对：C->T转换
# 3. 计算甲基化水平

def bisulfite_align(reads, ref):
    # 将参考基因组中所有C转换为T，得到T基因组
    ref_T = ref.replace('C', 'T')
    # 将reads中C转换为T
    reads_T = [r.replace('C', 'T') for r in reads]
    # 比对reads_T到ref_T
    alignments = align(reads_T, ref_T)
    return alignments

def methylation_level(alignments, ref):
    # 统计每个C位点的甲基化比例
    meth = {}
    for read, pos in alignments:
        if ref[pos] == 'C':
            if read[pos] == 'C':
                meth[pos] = meth.get(pos, [0,0])
                meth[pos][0] += 1  # 甲基化
            elif read[pos] == 'T':
                meth[pos] = meth.get(pos, [0,0])
                meth[pos][1] += 1  # 未甲基化

# Hi-C 数据处理核心步骤（Python 风格伪代码）
# 输入: paired-end reads (R1.fq, R2.fq), reference genome
# 输出: contact matrix

def hic_pipeline(r1, r2, genome):
    # 1. 比对：将双端 reads 比对到参考基因组
    alignments = bowtie2(r1, r2, genome)   # O(N * L), N=reads数, L=读长

    # 2. 过滤：去除未比对、多比对、PCR 重复
    valid_pairs = filter_pairs(alignments)  # O(N)

    # 3. 酶切位点附近重新定位（5' 端）
    pairs = re_align_to_restriction_sites(valid_pairs)  # O(N)

    # 4. 按基因组 bin 划分（如 1Mb / 100kb / 10kb）
    bins = assign_bins(pairs, bin_size=1_000_000)  # O(N)

    # 5. 构建接触矩阵
    matrix = build_contact_matrix(bins)  # O(N + B^2), B=bin数

    # 6. 归一化（ICE / KR）
    matrix_norm = ice_normalize(matrix)  # O(B^3) 迭代
    return matrix_norm

# 极简示例：3 个 bin 的接触矩阵
# bin1-bin1: 10, bin1-bin2: 3, bin1-bin3: 1
# bin2-bin2: 8,  bin2-bin3: 2
# bin3-bin3: 5
# matrix = [[10,3,1],[3,8,2],[1,2,5]]

import numpy as np

def build_contact_matrix(pairs, n_bins):
m = np.zeros((n_bins, n_bins), dtype=float)
for i, j, c in pairs:
m[i, j] += c
if i != j:
m[j, i] += c
return m

def ice_normalize(matrix, iterations=50, tol=1e-6):
m = matrix.astype(float).copy()
mask = m.sum(axis=1) > 0
for _ in range(iterations):
row_sum = m.sum(axis=1)
scale = np.ones_like(row_sum)
scale[mask] = row_sum[mask].mean() / row_sum[mask]
new_m = m * scale[:, None] * scale[None, :]
if np.max(np.abs(new_m - m)) < tol:
m = new_m
break
m = new_m
return m

pairs = [(0,0,10),(0,1,3),(0,2,1),(1,1,8),(1,2,2),(2,2,5)]
mat = build_contact_matrix(pairs, 3)
print('raw matrix:')
print(mat)
norm = ice_normalize(mat)
print('ICE normalized:')
print(np.round(norm, 4))
print('row sums after ICE:', np.round(norm.sum(axis=1), 4))

import numpy as np
from sklearn.cluster import SpectralClustering

def cluster_contigs_by_hic(contact_matrix, n_clusters):
    """
    contact_matrix: contig 间 Hi-C 接触矩阵 (C x C)
    n_clusters: 预期物种数
    返回: 每个 contig 的物种标签
    """
    # 归一化
    norm = contact_matrix / contact_matrix.sum(axis=1, keepdims=True)
    # 谱聚类
    sc = SpectralClustering(n_clusters=n_clusters, affinity='precomputed')
    labels = sc.fit_predict(norm)
    return labels

# 时间复杂度：谱聚类 O(C^3)，C=contig 数
